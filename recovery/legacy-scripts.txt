    pageLanguage: 'en',
    includedLanguages: 'en,am,om',

function triggerGTranslate(langCode) {
  /* Map our codes to Google's lang codes */
  var map = { en: 'en', am: 'am', om: 'om' };
  var target = map[langCode] || langCode;
  if (target === 'en') {
    /* Restore original — reload without cookie */
    var c = document.cookie.split(';');
    for (var i = 0; i < c.length; i++) {
      var name = c[i].trim().split('=')[0];
      if (name === 'googtrans') {
        document.cookie = 'googtrans=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;';
        document.cookie = 'googtrans=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/; domain=' + location.hostname + ';';
      }
    }
    location.reload();
    return;
  }
  /* Set the googtrans cookie then reload */
  var val = '/en/' + target;
  document.cookie = 'googtrans=' + val + '; path=/;';
  document.cookie = 'googtrans=' + val + '; path=/; domain=' + location.hostname + ';';
  location.reload();
}

/* ── Telegram iframe load handler ── */
function tgIframeLoaded() {
  const iframe   = document.getElementById('tg-iframe');
  const skeleton = document.getElementById('tg-skeleton');
  const fallback = document.getElementById('tg-fallback');

  if (!iframe) return;

  /* Fade out skeleton, fade in iframe */
  if (skeleton) {
    skeleton.style.transition = 'opacity 0.4s ease';
    skeleton.style.opacity = '0';
    setTimeout(() => { skeleton.style.display = 'none'; }, 420);
  }

  iframe.style.display = 'block';
  iframe.style.opacity = '0';
  iframe.style.transition = 'opacity 0.5s ease';
  setTimeout(() => { iframe.style.opacity = '1'; }, 50);
}

/* ── Fallback: if iframe doesn't load in 6s, show card posts ── */
setTimeout(function() {
  const iframe   = document.getElementById('tg-iframe');
  const skeleton = document.getElementById('tg-skeleton');
  const fallback = document.getElementById('tg-fallback');
  if (!iframe || iframe.style.display === 'block') return;

  /* iframe blocked or slow — show fallback cards */
  if (skeleton) skeleton.style.display = 'none';
  if (iframe)   iframe.style.display   = 'none';
  if (fallback) {
    fallback.style.display = 'flex';
    /* stagger-in each post */
    fallback.querySelectorAll('.tg-post').forEach((p, i) => {
      p.style.opacity = '0';
      p.style.transform = 'translateY(14px)';
      p.style.transition = `opacity 0.4s ease ${i * 0.09}s, transform 0.4s ease ${i * 0.09}s`;
      requestAnimationFrame(() => {
        p.style.opacity = '1';
        p.style.transform = 'translateY(0)';
      });
    });
  }
}, 6000);

(function() {
  // Light is now default (marketplace style). Dark adds .light class.
  var saved = localStorage.getItem('beki-theme');
  var isDark = saved === 'dark';
  if (isDark) document.documentElement.classList.add('light');
})();

function toggleTheme() {
  var isDark = document.documentElement.classList.contains('light');
  if (isDark) {
    document.documentElement.classList.remove('light');
    localStorage.setItem('beki-theme', 'light');
  } else {
    document.documentElement.classList.add('light');
    localStorage.setItem('beki-theme', 'dark');
  }
  syncSelectOptions();
}

function syncSelectOptions() {
  var isDark = document.documentElement.classList.contains('light');
  var bg = isDark ? '#1C2333' : '#FFFFFF';
  var clr = isDark ? '#F0F4FF' : '#1A2033';
  var opts = document.querySelectorAll('#langDropdown option');
  if (opts.length) {
    opts.forEach(function(o) { o.style.background = bg; o.style.color = clr; });
  }
}

/* ─── HERO SLIDESHOW — interactive, touch-friendly ─── */
const slides = document.querySelectorAll('.hero-slide');
const dots   = document.querySelectorAll('.hi-dot');
let cur = 0, slideTimer, isTransitioning = false;

function goToSlide(n) {
  if (isTransitioning || n === cur) return;
  isTransitioning = true;
  slides[cur].classList.remove('active');
  dots[cur].classList.remove('active');
  cur = (n + slides.length) % slides.length;
  slides[cur].classList.add('active');
  dots[cur].classList.add('active');
  /* allow next transition after crossfade completes */
  setTimeout(() => { isTransitioning = false; }, 1700);
}

function nextSlide() { goToSlide(cur + 1); }
function prevSlide() { goToSlide(cur - 1); }

function startTimer() {
  clearInterval(slideTimer);
  slideTimer = setInterval(nextSlide, 6000);
}
startTimer();

// Arrow buttons removed

/* Dot clicks reset timer */

  dot.addEventListener('click', () => { goToSlide(i); startTimer(); });
});

/* Touch / swipe support */
const slideshow = document.getElementById('heroSlideshow');
let touchStartX = 0, touchStartY = 0;
slideshow.addEventListener('touchstart', e => {
  touchStartX = e.touches[0].clientX;
  touchStartY = e.touches[0].clientY;
}, { passive: true });
slideshow.addEventListener('touchend', e => {
  const dx = e.changedTouches[0].clientX - touchStartX;
  const dy = e.changedTouches[0].clientY - touchStartY;
  /* only trigger if horizontal swipe > 40px and not a scroll */
  if (Math.abs(dx) > 40 && Math.abs(dx) > Math.abs(dy) * 1.5) {
    dx < 0 ? nextSlide() : prevSlide();
    startTimer();
  }
}, { passive: true });

/* Pause autoplay while user hovers (desktop) */
slideshow.addEventListener('mouseenter', () => clearInterval(slideTimer));
slideshow.addEventListener('mouseleave', startTimer);

/* Pause when tab is hidden to save battery */
document.addEventListener('visibilitychange', () => {
  document.hidden ? clearInterval(slideTimer) : startTimer();
});

function getActivePanel() {
  return document.querySelector('.cars-panel.active');
}
function scrollCarousel(dir) {
  const panel = getActivePanel();
  if (!panel) return;
  const cardW = panel.querySelector('.car-card') ? panel.querySelector('.car-card').offsetWidth + 1 : 302;
  panel.scrollBy({ left: dir * cardW * 2, behavior: 'smooth' });
  setTimeout(updateCarouselArrows, 350);
}
function updateCarouselArrows() {
  const panel = getActivePanel();
  const prev = document.getElementById('carouselPrev');
  const next = document.getElementById('carouselNext');
  if (!panel || !prev || !next) return;
  prev.classList.toggle('hidden', panel.scrollLeft <= 4);
  next.classList.toggle('hidden', panel.scrollLeft + panel.clientWidth >= panel.scrollWidth - 4);
}
document.addEventListener('DOMContentLoaded', () => {
  const panel = getActivePanel();
  if (panel) {
    panel.addEventListener('scroll', updateCarouselArrows, { passive: true });
    updateCarouselArrows();
  }
});

(function(){

  /* ── All searchable data — mirrors CARS array + site sections ── */
  const GS_CARS = [];  /* populated after CARS array is defined */
  const GS_SECTIONS = [
    { label:'Live Inventory',   icon:'fa-car',         href:'#new-cars',          desc:'Browse all cars for sale & rent' },
    { label:'Services',         icon:'fa-star',        href:'#services',          desc:'Import sourcing, rental & brokerage' },
    { label:'Telegram Updates', icon:'fa-brands fa-telegram', href:'#telegram-updates', desc:'Latest arrivals & price drops' },
    { label:'Our Showroom',     icon:'fa-map-pin',     href:'#showroom',          desc:'Visit us in Addis Ababa' },
    { label:'Contact / Inquire',icon:'fa-paper-plane', href:'#contact',           desc:'Send inquiry or get a callback' },
  ];

  const SUGGESTIONS = ['Hyundai Tucson','Toyota Vitz','Suzuki Dzire','SUV','For Sale','For Rent','Automatic','Petrol','EV'];

  let gsFocusIdx = -1;
  let gsOpen = false;
  let gsClickingDropdown = false;

  /* ── DOM refs (resolved after DOM ready) ── */
  function $(id){ return document.getElementById(id); }

  /* ── Populate GS_CARS from the live CARS array ── */
  function gsSync(){
    GS_CARS.length = 0;
    /* CARS is defined in the inventory IIFE scope — read via DOM data */
    const stored = window.__GS_CARS__;
    if (stored) stored.forEach(c => GS_CARS.push(c));
  }

  /* ── Highlight match in string ── */
  function gsHighlight(str, q){
    if (!q) return str;
    const re = new RegExp('(' + q.replace(/[.*+?^${}()|[\]\\]/g,'\\$&') + ')', 'gi');
    return String(str).replace(re, '<span class="gs-highlight">$1</span>');
  }

  /* ── Search logic ── */
  function gsSearch(q){
    q = q.trim().toLowerCase();
    if (!q) return [];
    return GS_CARS.filter(c => {
      const blob = [c.make, c.model, String(c.year), c.fuel, c.transmission,
        c.bodyType||'', c.origin||'', c.condition||'', c.priceLabel,
        c.category==='sale'?'for sale':'for rent',
        ...(c.features||[])
      ].join(' ').toLowerCase();
      return blob.includes(q);
    });
  }

  /* ── Open / close dropdown ── */
  function gsShowDropdown(){
    const dd = $('gs-dropdown');
    if (!dd) return;
    dd.style.display = 'block';
    gsOpen = true;
  }
  function gsHideDropdown(){
    const dd = $('gs-dropdown');
    if (!dd) return;
    dd.style.display = 'none';
    gsOpen = false;
    gsFocusIdx = -1;
  }

  /* ── Render suggestions (shown when input empty) ── */
  function gsRenderSuggestions(){
    $('gs-suggestions').style.display = 'block';
    $('gs-results-list').style.display = 'none';
    $('gs-no-results').style.display   = 'none';
    $('gs-result-label').textContent   = 'Quick Search';
    $('gs-see-all').style.display      = 'none';

    /* Pills */
    $('gs-suggestion-pills').innerHTML = SUGGESTIONS.map(s =>
      `<button class="gs-suggestion-pill" onclick="gsFillSuggestion('${s}')">${s}</button>`
    ).join('');

    /* Quick nav links */
    const saleCount = GS_CARS.filter(c=>c.category==='sale').length;
    const rentCount = GS_CARS.filter(c=>c.category==='rent').length;
    $('gs-quick-links').innerHTML = [
      {icon:'fa-tag',   label:'Cars for Sale',  count:`${saleCount} listings`, href:'#new-cars', cat:'sale'},
      {icon:'fa-key',   label:'Cars for Rent',  count:`${rentCount} listings`, href:'#new-cars', cat:'rent'},
      {icon:'fa-bolt',  label:'Urgent Listings',count:GS_CARS.filter(c=>c.urgent).length+' car', href:'#new-cars', cat:'all'},
      {icon:'fa-brands fa-telegram', label:'Telegram Channel', count:'Live updates', href:'https://t.me/bekiautocar', ext:true},
    ].map(l => `
      <${l.ext?`a href="${l.href}" target="_blank"`:`a href="${l.href}"`} class="gs-quick-link" onclick="${l.cat?`gsFilterAndGo('${l.cat}')`:'gsHideDropdown()'}">
        <span class="gs-quick-link-icon"><i class="fa-solid ${l.icon}"></i></span>
        <span class="gs-quick-link-label">${l.label}</span>
        <span class="gs-quick-link-count">${l.count}</span>
      </a>`).join('');
  }

  /* ── Render search results ── */
  function gsRenderResults(results, q){
    $('gs-suggestions').style.display = 'none';
    $('gs-no-results').style.display  = 'none';
    $('gs-results-list').style.display = 'block';
    $('gs-result-label').textContent  = results.length + ' result' + (results.length!==1?'s':'') + ' found';
    $('gs-see-all').style.display     = results.length > 0 ? 'inline' : 'none';

    const list = $('gs-results-list');
    if (!results.length){
      $('gs-results-list').style.display = 'none';
      $('gs-no-results').style.display   = 'block';
      $('gs-result-label').textContent   = 'No results';
      $('gs-see-all').style.display      = 'none';
      return;
    }

    list.innerHTML = results.slice(0, 8).map((c, i) => {
      const name = gsHighlight(c.make + ' ' + c.model, q);
      const meta = [c.year, c.transmission, c.fuel, c.mileage].filter(Boolean).join(' · ');
      let badges = `<span class="gs-result-badge ${c.category==='sale'?'gs-badge-sale':'gs-badge-rent'}">${c.category==='sale'?'Sale':'Rent'}</span>`;
      if (c.urgent) badges += ` <span class="gs-result-badge gs-badge-urgent">Urgent</span>`;
      return `
        <div class="gs-result-item" data-idx="${i}" data-id="${c.id}"
          onmousedown="gsClickingDropdown=true"

          onmouseenter="gsSetFocus(${i})">

            <span style="display:flex;gap:4px;">${badges}</span>
          </div>
        </div>`;
    }).join('');

    if (results.length > 8){
      list.innerHTML += `<div style="text-align:center;padding:10px;"><a href="#new-cars" onclick="gsGoToSection()" style="font-size:0.72rem;font-weight:700;color:var(--gold);font-family:var(--sans);">+ ${results.length-8} more results →</a></div>`;
    }
  }

  /* ── Public handlers ── */
  window.gsHandleInput = function(val){
    const clearBtn = $('gs-clear');
    if (clearBtn) clearBtn.style.display = val ? 'block' : 'none';
    gsSync();
    gsShowDropdown();
    gsFocusIdx = -1;
    if (!val.trim()){
      gsRenderSuggestions();
    } else {
      const results = gsSearch(val);
      gsRenderResults(results, val.trim().toLowerCase());
    }
    /* also sync to inv-search if visible */
    const invSearch = $('inv-search');
    if (invSearch){ invSearch.value = val; if(window.invFilter) window.invFilter(); }
  };

  window.gsOnFocus = function(){
    gsSync();
    gsShowDropdown();
    const val = $('global-search-input').value;
    if (!val.trim()) gsRenderSuggestions();
    else { gsRenderResults(gsSearch(val), val.trim().toLowerCase()); }
    /* Expand box */
    $('global-search-box').style.borderColor = 'var(--gold)';
    $('global-search-box').style.boxShadow   = '0 0 0 3px rgba(201,162,39,0.12)';
  };

  window.gsOnBlur = function(){
    if (gsClickingDropdown){ gsClickingDropdown=false; return; }
    setTimeout(()=>{
      gsHideDropdown();
      $('global-search-box').style.borderColor = '';
      $('global-search-box').style.boxShadow   = '';
    }, 160);
  };

  window.gsClear = function(){
    const inp = $('global-search-input');
    inp.value = '';
    $('gs-clear').style.display = 'none';
    gsRenderSuggestions();
    inp.focus();
    const invSearch = $('inv-search');
    if (invSearch){ invSearch.value=''; if(window.invFilter) window.invFilter(); }
  };

  window.gsFillSuggestion = function(text){
    const inp = $('global-search-input');
    inp.value = text;
    $('gs-clear').style.display = 'block';
    gsSync();
    const results = gsSearch(text);
    gsRenderResults(results, text.toLowerCase());
    /* sync to inv section */
    const invSearch = $('inv-search');
    if (invSearch){ invSearch.value=text; if(window.invFilter) window.invFilter(); }
  };

  window.gsPickResult = function(id){
    gsHideDropdown();
    $('global-search-input').blur();
    /* scroll to inventory and open modal */
    const section = document.getElementById('new-cars');
    if (section){
      const hh = document.getElementById('site-header').offsetHeight;
      window.scrollTo({ top: section.getBoundingClientRect().top + window.scrollY - hh - 16, behavior:'smooth' });
    }
    setTimeout(()=>{ if(window.invOpenModal) window.invOpenModal(id); }, 420);
  };

  window.gsGoToSection = function(){
    gsHideDropdown();
    const q = $('global-search-input').value;
    const invSearch = $('inv-search');
    if (invSearch && q){ invSearch.value=q; if(window.invFilter) window.invFilter(); }
    const section = document.getElementById('new-cars');
    if (section){
      const hh = document.getElementById('site-header').offsetHeight;
      window.scrollTo({ top: section.getBoundingClientRect().top + window.scrollY - hh - 16, behavior:'smooth' });
    }
  };

  window.gsFilterAndGo = function(cat){
    gsHideDropdown();
    /* trigger pill tab */
    const pill = document.querySelector(`.inv-pill[data-cat="${cat}"]`);
    if (pill && window.invSetCat) window.invSetCat(pill, cat);
    const section = document.getElementById('new-cars');
    if (section){
      const hh = document.getElementById('site-header').offsetHeight;
      window.scrollTo({ top: section.getBoundingClientRect().top + window.scrollY - hh - 16, behavior:'smooth' });
    }
  };

  /* ── Keyboard nav ── */
  window.gsKeyNav = function(e){
    const items = document.querySelectorAll('.gs-result-item');
    if (!items.length) return;
    if (e.key === 'ArrowDown'){
      e.preventDefault();
      gsFocusIdx = Math.min(gsFocusIdx+1, items.length-1);
      gsSetFocus(gsFocusIdx);
    } else if (e.key === 'ArrowUp'){
      e.preventDefault();
      gsFocusIdx = Math.max(gsFocusIdx-1, 0);
      gsSetFocus(gsFocusIdx);
    } else if (e.key === 'Enter'){
      if (gsFocusIdx >= 0 && items[gsFocusIdx]){
        items[gsFocusIdx].click();
      } else {
        gsGoToSection();
      }
    } else if (e.key === 'Escape'){
      gsHideDropdown();
      $('global-search-input').blur();
    }
  };

  window.gsSetFocus = function(idx){
    document.querySelectorAll('.gs-result-item').forEach((el,i)=>{
      el.classList.toggle('gs-focused', i===idx);
    });
    gsFocusIdx = idx;
  };

  /* ── Dismiss on outside click ── */
  document.addEventListener('click', function(e){
    const wrap = $('global-search-wrap');
    if (wrap && !wrap.contains(e.target)) gsHideDropdown();
  });

  /* ── Init: render suggestions on load ── */
  document.addEventListener('DOMContentLoaded', function(){
    gsSync();
    gsRenderSuggestions();
  });

})();

    toastTimer = setTimeout(function(){ t.classList.remove('rf-show'); }, 3200);
  };

  function initials(name){
    if(!name) return 'U';
    var p = name.trim().split(/\s+/);
    return ((p[0]||'')[0] || '' ).concat((p[1]||'')[0] || '').toUpperCase() || 'U';
  }

  /* header account button reflects auth state */
  function paintAccountBtn(){
    var btn = document.getElementById('rf-account-btn');
    if(!btn) return;
    var u = load();
    if (u){
      btn.querySelector('.rf-avatar').textContent = initials(u.name || u.email);
      btn.querySelector('.rf-account-label').textContent = (u.name ? u.name.split(' ')[0] : 'Account');
    } else {
      btn.querySelector('.rf-avatar').innerHTML = '<i class="fa-solid fa-user" style="font-size:.6rem;"></i>';
      btn.querySelector('.rf-account-label').textContent = 'Sign In';
    }
  }

  var V = {
    name:  function(v){ return v.trim().length >= 2; },
    email: function(v){ return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v.trim()); },
    phone: function(v){ return /^(\+251|0)?9\d{8}$/.test(v.replace(/[\s-]/g,'')); },
    pass:  function(v){ return v.length >= 6; }
  };
  var FIELD = { name:'rf-f-name', email:'rf-f-email', phone:'rf-f-phone', pass:'rf-f-pass' };
  var INPUT = { name:'rf-i-name', email:'rf-i-email', phone:'rf-i-phone', pass:'rf-i-pass' };

  function setValidity(field, ok, touched){
    var wrap = document.getElementById(FIELD[field]);
    if(!wrap) return;
    wrap.classList.toggle('rf-valid', ok);
    wrap.classList.toggle('rf-invalid', touched && !ok);
  }

  window.rfAuth = {
    open: function(carId){
      state.pendingCarId = carId || null;
      this.applyMode();
      var ov = document.getElementById('rf-auth-overlay');
      ov.classList.add('rf-show');
      document.body.style.overflow = 'hidden';
      var u = load();
      if (u){ this.showDashboard(u); }
      else {
        document.getElementById('rf-dash').classList.remove('rf-show');
        document.getElementById('rf-auth-form').style.display = 'block';
        // autofill remembered email
        var last = localStorage.getItem('rf_last_email');
        if (last){ document.getElementById('rf-i-email').value = last; }
        setTimeout(function(){
          var f = document.getElementById(u ? null : 'rf-i-email');
          if (f) f.focus();
        }, 320);
      }
    },
    close: function(){
      document.getElementById('rf-auth-overlay').classList.remove('rf-show');
      document.body.style.overflow = '';
    },
    backdrop: function(e){ if (e.target === document.getElementById('rf-auth-overlay')) this.close(); },

    role: function(r){
      state.role = r;
      document.getElementById('rf-role-buyer').classList.toggle('rf-active', r==='buyer');
      document.getElementById('rf-role-seller').classList.toggle('rf-active', r==='seller');
      document.getElementById('rf-role-buyer').setAttribute('aria-pressed', r==='buyer');
      document.getElementById('rf-role-seller').setAttribute('aria-pressed', r==='seller');
    },

    toggleMode: function(){
      state.mode = (state.mode === 'signin') ? 'signup' : 'signin';
      this.applyMode();
    },
    applyMode: function(){
      var signup = state.mode === 'signup';
      document.getElementById('rf-auth-title').textContent = signup ? 'Create your account' : 'Welcome back';
      document.getElementById('rf-auth-sub').textContent = signup
        ? 'Join free to unlock prices, full specs & direct seller contact.'
        : 'Sign in to view full vehicle details, prices & seller contact.';
      document.getElementById('rf-role-row').style.display = signup ? 'flex' : 'none';
      document.getElementById('rf-f-name').style.display   = signup ? 'block' : 'none';
      document.getElementById('rf-f-phone').style.display  = signup ? 'block' : 'none';
      document.getElementById('rf-consent').style.display  = signup ? 'block' : 'none';
      document.getElementById('rf-i-pass').setAttribute('autocomplete', signup ? 'new-password' : 'current-password');
      document.querySelector('#rf-submit .rf-btn-text').textContent = signup ? 'Create Account' : 'Sign In';
      document.getElementById('rf-switch-text').textContent = signup ? 'Already have an account?' : 'New to Beki AutoCar?';
      document.getElementById('rf-switch-btn').textContent  = signup ? 'Sign in instead' : 'Create an account';
      // reset validation visuals
      ['name','email','phone','pass'].forEach(function(f){
        var w = document.getElementById(FIELD[f]); if(w) w.classList.remove('rf-valid','rf-invalid');
      });
    },

    live: function(field){
      var v = document.getElementById(INPUT[field]).value;
      setValidity(field, V[field](v), v.length > 0);
    },

    activeFields: function(){
      return state.mode === 'signup' ? ['name','email','phone','pass'] : ['email','pass'];
    },

    submit: function(){
      var self = this, ok = true;
      this.activeFields().forEach(function(f){
        var valid = V[f](document.getElementById(INPUT[f]).value);
        setValidity(f, valid, true);
        if (!valid) ok = false;
      });
      if (!ok){
        var firstBad = document.querySelector('.rf-field.rf-invalid input');
        if (firstBad) firstBad.focus();
        return;
      }
      var btn = document.getElementById('rf-submit');
      btn.classList.add('rf-loading'); btn.disabled = true;
      var email = document.getElementById('rf-i-email').value.trim();
      var user = {
        name:  document.getElementById('rf-i-name').value.trim() || email.split('@')[0],
        email: email,
        phone: document.getElementById('rf-i-phone').value.trim(),
        role:  state.mode === 'signup' ? state.role : (load() && load().role) || 'buyer',
        since: Date.now(),
        saved: [], views: 0, inquiries: 0
      };
      // simulate async auth
      setTimeout(function(){
        btn.classList.remove('rf-loading'); btn.disabled = false;
        save(user);
        try{ localStorage.setItem('rf_last_email', email); }catch(e){}
        paintAccountBtn();
        rfToast(state.mode==='signup' ? 'Account created — welcome!' : 'Signed in successfully', 'fa-circle-check');
        self.showDashboard(user);
        // if a car detail was pending, reopen it now unlocked
        if (state.pendingCarId != null && typeof window.invOpenModal === 'function'){
          var pend = state.pendingCarId; state.pendingCarId = null;
          self.close();
          setTimeout(function(){ window.invOpenModal(pend); }, 260);
        }
      }, reduce ? 100 : 850);
    },

    showDashboard: function(u){
      document.getElementById('rf-auth-form').style.display = 'none';
      document.getElementById('rf-role-row').style.display = 'none';
      document.getElementById('rf-consent').style.display = 'none';
      document.querySelector('.rf-switch').style.display = 'none';
      document.getElementById('rf-auth-title').textContent = 'Your account';
      document.getElementById('rf-auth-sub').textContent = 'Manage your Beki AutoCar activity.';
      document.getElementById('rf-dash-name').textContent = u.name || u.email;
      var roleEl = document.getElementById('rf-dash-role');
      roleEl.innerHTML = '<i class="fa-solid fa-circle-check"></i> ' + (u.role === 'seller' ? 'Seller' : 'Buyer');
      document.getElementById('rf-stat-saved').textContent = (u.saved||[]).length;
      document.getElementById('rf-stat-views').textContent = u.views || 0;
      document.getElementById('rf-stat-inq').textContent   = u.inquiries || 0;
      // role-specific quick actions
      var acts = (u.role === 'seller') ? [
        ['fa-plus','List a vehicle','Post your car to Beki AutoCar'],
        ['fa-chart-line','My listings','Track views & inquiries'],
        ['fa-headset','Seller support','Talk to a broker on Telegram']
      ] : [
        ['fa-heart','Saved vehicles','Cars you bookmarked'],
        ['fa-bell','Price alerts','Get notified on drops'],
        ['fa-headset','Buyer support','Chat with Beki on Telegram']
      ];
      document.getElementById('rf-dash-actions').innerHTML = acts.map(function(a){
        return '<button class="rf-dash-act" onclick="rfAuth.dashAction(\''+a[1]+'\')">'
             + '<i class="fa-solid '+a[0]+'"></i><div><b>'+a[1]+'</b><small>'+a[2]+'</small></div>'
             + '<i class="fa-solid fa-chevron-right" style="color:var(--text-muted);font-size:.7rem;"></i></button>';
      }).join('');
      document.getElementById('rf-dash').classList.add('rf-show');
    },

    dashAction: function(label){
      if (/support/i.test(label)){ window.open('https://t.me/Beckyyyyyy','_blank'); return; }
      rfToast(label + ' — coming soon', 'fa-circle-info');
    },

    signOut: function(){
      clear();
      paintAccountBtn();
      document.getElementById('rf-dash').classList.remove('rf-show');
      document.querySelector('.rf-switch').style.display = 'block';
      state.mode = 'signin'; this.applyMode();
      document.getElementById('rf-auth-form').style.display = 'block';
      rfToast('Signed out', 'fa-circle-info');
      this.close();
    },

    isAuthed: function(){ return !!load(); },
    bumpView: function(){
      var u = load(); if(!u) return;
      u.views = (u.views||0) + 1; save(u);
    }
  };

  /* ── 6 · INJECT ACCOUNT BUTTON into header-right ── */
  function injectAccountBtn(){
    var hr = document.querySelector('.header-right');
    if (!hr || document.getElementById('rf-account-btn')) return;
    var btn = document.createElement('button');
    btn.id = 'rf-account-btn';
    btn.className = 'rf-account-btn';
    btn.type = 'button';
    btn.setAttribute('aria-label','Account / sign in');
    btn.onclick = function(){ rfAuth.open(); };
    btn.innerHTML = '<span class="rf-avatar"></span><span class="rf-account-label">Sign In</span>';
    var menuBtn = document.getElementById('mob-menu-btn');
    if (menuBtn) hr.insertBefore(btn, menuBtn); else hr.appendChild(btn);
    paintAccountBtn();
  }

  /* ── 7 · DETAIL GATE: wrap invOpenModal so prices/specs/contact
        require an account. Preview (image, title, badges) stays open
        to entice; full detail prompts sign-up. ── */
  function installGate(){
    if (typeof window.invOpenModal !== 'function'){ setTimeout(installGate, 200); return; }
    if (window.__rfGated) return; window.__rfGated = true;
    var original = window.invOpenModal;
    window.invOpenModal = function(id){
      original(id);                       // render the modal as usual
      var authed = rfAuth.isAuthed();
      // locate price/specs/contact blocks inside the rendered modal
      var price = document.getElementById('invModalPrice');
      var specs = document.getElementById('invModalSpecs');
      var tg    = document.getElementById('invModalTgNote');
      var feat  = document.getElementById('invModalFeatures');
      // remove any prior gate
      var oldGate = document.getElementById('rf-detail-gate');
      if (oldGate) oldGate.remove();

      if (authed){
        rfAuth.bumpView();
        [price,specs,tg,feat].forEach(function(el){ if(el) el.style.filter=''; });
        return;
      }
      // NOT signed in → blur the sensitive blocks + inject gate CTA
      [price,specs,feat].forEach(function(el){
        if(el){ el.style.filter='blur(7px)'; el.style.userSelect='none'; el.style.pointerEvents='none'; }
      });
      if (tg) tg.style.display = 'none';
      var gate = document.createElement('div');
      gate.id = 'rf-detail-gate'; gate.className = 'rf-gate';
      gate.innerHTML =
        '<i class="fa-solid fa-lock rf-lock"></i>'
       +'<h4>See the full details</h4>'
       +'<p>Create a free account to unlock the price, full specs, features and seller contact for this vehicle.</p>'
       +'<button class="rf-btn" style="max-width:260px;margin:0 auto;" onclick="rfAuth.signupFor('+id+')">'
       +'<span class="rf-btn-text"><i class="fa-solid fa-unlock"></i>&nbsp; Unlock free</span>'
       +'<span class="rf-pl-ring" aria-hidden="true"></span></button>';
      if (price && price.parentNode){
        price.parentNode.insertBefore(gate, price.nextSibling);
      }
    };
  }
  rfAuth.signupFor = function(id){ state.mode='signup'; rfAuth.open(id); };

  /* ── 8 · keyboard: Esc closes auth, role toggles via Enter/Space ── */
  document.addEventListener('keydown', function(e){
    if (e.key === 'Escape'){
      var ov = document.getElementById('rf-auth-overlay');
      if (ov && ov.classList.contains('rf-show')) rfAuth.close();
    }
  });
  ['rf-role-buyer','rf-role-seller'].forEach(function(rid){
    document.addEventListener('keydown', function(e){
      if (e.target && e.target.id === rid && (e.key==='Enter'||e.key===' ')){
        e.preventDefault(); e.target.click();
      }
    });
  });

  /* ── INIT ── */
  document.addEventListener('DOMContentLoaded', function(){
    injectAccountBtn();
    installGate();
  });
  // header may already exist if DOM parsed; run now too
  injectAccountBtn(); installGate();
})();

  }

  /* Stat counter animation */

  var statObs = new IntersectionObserver(function(entries){
    entries.forEach(function(e){

        statDone = true;
        document.querySelectorAll('.hero-stat-num').forEach(function(el){
          var txt = el.textContent.trim();
          var num = parseInt(txt.replace(/[^0-9]/g,''),10);
          if(isNaN(num)||num===0) return;
          var suffix = txt.replace(/[0-9]/g,'');
          var s = 0, step = num/70;
          var t = setInterval(function(){s=Math.min(s+step,num);el.textContent=Math.floor(s)+suffix;if(s>=num)clearInterval(t);},18);
        });
      }
